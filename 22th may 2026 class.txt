use vitcampus;

create table department(
did int primary key,
dname varchar(25) not null unique);

insert into department values(101,"cse"),(102,"ece");
insert into department values(103,"civil"),(106,"aero"),(107,"mechanical");

select * from department;

select * from department order by did asc;

create table hod(
hid int primary key,
hodname varchar(20) not null unique);

insert into hod values(101,"arun"),(102,"jayanth");
insert into hod values(104,"karthik"),(105,"murali");

select *from hod;

select * from department cross join hod order by did;

select * from department inner join hod where department.did=hod.hid;


select * from department natural join hod;#same as cross join but check alter


select * from department left outer join hod on(department.did=hod.hid) order by did asc;

select * from department full join hod on (department.did=hod.hid);
